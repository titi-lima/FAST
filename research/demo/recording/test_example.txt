
import pytest
from pricing import shipping_cost


@pytest.mark.parametrize("amount, express, expected", [
    (0, False, 5),
    (10, False, 5),
    (49, False, 5),
    (50, False, 0),
    (100, False, 0),
    (0, True, 12),
    (10, True, 12),
    (49, True, 12),
    (50, True, 12),
    (100, True, 12),
])
def test_shipping(amount, express, expected):
    assert shipping_cost(amount, express) == expected
